# --- Computer vision ------------------------------------------------------
# Two constraints on this line, and both matter.
#
# 1. It must be opencv-CONTRIB-python, never plain opencv-python. The LBPH
#    recognizer used for identity verification lives in cv2.face, which ships
#    only in the contrib build. Both packages install a module called cv2, so
#    if the plain build is present as well it can shadow the contrib build:
#    the import still succeeds, cv2.face is simply absent, and verification
#    silently falls back to a much weaker correlation match. If in doubt:
#        pip uninstall -y opencv-python opencv-python-headless
#
# 2. It must stay below 5.0. OpenCV 5 removed the Haar cascade XML files that
#    used to ship inside the wheel, and face_engine.py loads its face and eye
#    detectors from cv2.data.haarcascades. On OpenCV 5 that directory is empty,
#    both cascades fail to load, and no face is ever detected. cv2.face itself
#    still exists in 5.x - it is the detector, not the recognizer, that breaks.
opencv-contrib-python>=4.10.0,<5.0
numpy>=1.26.0,<3.0

# --- Geofencing -----------------------------------------------------------
geopy>=2.4.1,<3.0

# --- Worker identity cards -------------------------------------------------

# install small. python-barcode draws Code 128, qrcode draws QR.

qrcode>=7.4,<9.0
